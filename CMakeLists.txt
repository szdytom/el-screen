cmake_minimum_required(VERSION 3.16)
project(el_screen LANGUAGES C)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)

set(tuning_path "${CMAKE_CURRENT_SOURCE_DIR}/src/tuning.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${tuning_path}")
file(READ "${tuning_path}" tuning)

function(embed_shader variable name)
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${name}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source)
    string(FIND "${source}" "\n" version_end)
    math(EXPR body_start "${version_end} + 1")
    string(SUBSTRING "${source}" 0 "${body_start}" version)
    string(SUBSTRING "${source}" "${body_start}" -1 body)
    set(source "${version}${tuning}\n#line 2\n${body}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shaders/${name}" "${source}")
    string(REPLACE "\\" "\\\\" source "${source}")
    string(REPLACE "\"" "\\\"" source "${source}")
    string(REPLACE "\n" "\\n\"\n\"" source "${source}")
    set(${variable} "\"${source}\"" PARENT_SCOPE)
endfunction()

embed_shader(VERTEX_SHADER fullscreen.vert)
embed_shader(DRIVE_SHADER drive.frag)
embed_shader(BLOOM_SHADER bloom.frag)
embed_shader(PANEL_SHADER panel.frag)
configure_file(src/shader_sources.h.in shader_sources.h @ONLY)

add_executable(el-screen src/main.c src/display.c src/pages.c src/raster.c)
set_target_properties(el-screen PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(el-screen PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(el-screen PRIVATE PkgConfig::SDL3 PkgConfig::GLES m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(el-screen PRIVATE -Wall -Wextra -Wshadow)
endif()
